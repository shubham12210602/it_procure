Production Relational Database Schema & Backend Architecture

PostgreSQL / MySQL compatible DDL, Foreign Key relationships, and Email threading design.

schema.sql Includes Indexes, Foreign Keys, Audit logs & Email headers 
-- ========================================================
-- IT-ProcureDesk: Production PostgreSQL / MySQL Schema
-- ========================================================

-- 1. TICKETS TABLE: Core Procurement Requisitions
CREATE TABLE tickets (
    id VARCHAR(32) PRIMARY KEY,                   -- e.g. 'IT-1042'
    requester_name VARCHAR(150) NOT NULL,
    requester_email VARCHAR(255) NOT NULL,
    department VARCHAR(100) NOT NULL,
    item_name VARCHAR(255) NOT NULL,
    category VARCHAR(60) NOT NULL,                -- 'Laptop', 'Software', 'Monitor'
    estimated_cost DECIMAL(12, 2) NOT NULL,
    currency VARCHAR(3) DEFAULT 'USD',
    justification TEXT NOT NULL,
    vendor_link TEXT,
    priority VARCHAR(20) DEFAULT 'Medium',        -- 'Low', 'Medium', 'High', 'Urgent'
    status VARCHAR(30) DEFAULT 'New Request',     -- 'New Request', 'In Approval', 'Approved', 'Ordered', 'Fulfilled', 'Rejected'
    assigned_to VARCHAR(150),
    source VARCHAR(20) DEFAULT 'Portal',          -- 'Portal', 'Gmail', 'Slack'
    
    -- Gmail Inbound Threading Headers
    gmail_message_id VARCHAR(255),               -- RFC 2822 Message-ID
    gmail_thread_id VARCHAR(255),                -- Google Gmail Thread ID
    
    -- Procurement & Shipping info
    po_number VARCHAR(50),
    vendor_name VARCHAR(120),
    tracking_number VARCHAR(100),
    estimated_delivery DATE,
    
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2. APPROVALS TABLE: Multi-level Sign-offs
CREATE TABLE ticket_approvals (
    id SERIAL PRIMARY KEY,
    ticket_id VARCHAR(32) REFERENCES tickets(id) ON DELETE CASCADE,
    approval_type VARCHAR(50) NOT NULL,           -- 'Manager', 'Finance', 'Security_Infosec'
    approver_email VARCHAR(255) NOT NULL,
    status VARCHAR(20) DEFAULT 'Pending',         -- 'Pending', 'Approved', 'Rejected'
    comments TEXT,
    approved_at TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 3. COMMENTS & GMAIL THREAD LOGS TABLE
CREATE TABLE ticket_comments (
    id SERIAL PRIMARY KEY,
    ticket_id VARCHAR(32) REFERENCES tickets(id) ON DELETE CASCADE,
    author_name VARCHAR(150) NOT NULL,
    author_email VARCHAR(255) NOT NULL,
    is_internal_note BOOLEAN DEFAULT FALSE,       -- Internal IT only vs sent to user
    is_email_reply BOOLEAN DEFAULT FALSE,         -- Sent via Gmail SMTP
    content TEXT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 4. ATTACHMENTS TABLE (Quotes, Invoices, Receipts)
CREATE TABLE ticket_attachments (
    id SERIAL PRIMARY KEY,
    ticket_id VARCHAR(32) REFERENCES tickets(id) ON DELETE CASCADE,
    file_name VARCHAR(255) NOT NULL,
    file_size_kb INT NOT NULL,
    storage_url TEXT NOT NULL,                    -- S3 or GCS Signed URL
    uploaded_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 5. AUDIT LOGS FOR SOC2 / ISO27001 COMPLIANCE
CREATE TABLE audit_logs (
    id BIGSERIAL PRIMARY KEY,
    ticket_id VARCHAR(32) REFERENCES tickets(id) ON DELETE CASCADE,
    actor_email VARCHAR(255) NOT NULL,
    action_type VARCHAR(100) NOT NULL,
    old_value JSONB,
    new_value JSONB,
    timestamp TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- Performance Indexes
CREATE INDEX idx_tickets_status ON tickets(status);
CREATE INDEX idx_tickets_requester ON tickets(requester_email);
CREATE INDEX idx_tickets_gmail_thread ON tickets(gmail_thread_id);
CREATE INDEX idx_approvals_ticket ON ticket_approvals(ticket_id);

Gmail Integration Pattern

To connect your actual Google Workspace mailbox:

  1. Create a Service Account or OAuth App in Google Cloud Console.
  2. Enable Gmail API with scope gmail.readonly & gmail.send.
  3. Subscribe via Gmail Push Notifications (`users.watch`) pushing to a Google Pub/Sub topic.
  4. When incoming email hits `it-procure@company.com`, your webhook parses headers (Subject, From, In-Reply-To).
  5. If Subject has [IT-XXXX], append to existing ticket conversation; otherwise, create a new record.
